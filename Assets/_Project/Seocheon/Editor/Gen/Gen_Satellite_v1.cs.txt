//  출력  Assets/_Project/Seocheon/Art/Sky/T_Satellite_v1.png
//        RGBA · 256 x 256 · sRGB on · Alpha Is Transparency on · mipmap on
//        Android ASTC 6x6
//  RGB   회색 바탕 + 저대비 얼룩(밸류노이즈 fBm 3옥타브) + 원형 딤플(크레이터풍)
//  A     원판 마스크. 가장자리 1.5px smoothstep
//
//  ★목표 5x5 국소 SD 정규화 0.020 ~ 0.035  (행성 0.0160 보다 높게)
//    달은 얼룩덜룩한 것이 정상이다. 너무 매끈하면 또 유리구슬이 된다 (함정 ③)
//  ★진폭은 탐색하지 않는다. 단위 진폭으로 한 번 굽고 SD 를 재서 선형 보정 후 다시 굽는다
//    (SD 는 진폭에 선형이므로 1회 보정으로 정확히 맞는다)
//
//  ★CodeDom = C# 6. 로컬 함수 금지 -> System.Func. 한글 식별자 금지.
//  ★픽셀 루프 안에서 Unity 네이티브 API 호출 금지.
// ============================================================================

const int S = 256;
const string DST = "Assets/_Project/Seocheon/Art/Sky/T_Satellite_v1.png";
const double BASE   = 0.72;     // 바탕 밝기 (표시/sRGB 공간)
const double SDGOAL = 0.027;    // 국소 SD 정규화 목표 (0.020~0.035 한가운데)
const double DISCR  = 124.0;    // 원판 반경 px (256 에서 4px 여백)
const double EDGE   = 1.5;      // 가장자리 전이 폭 px

System.Func<double,double> SS = delegate(double q){ if(q<0)q=0; else if(q>1)q=1; return q*q*(3.0-2.0*q); };

// ---- 2D 값 잡음 (정수 해시, smoothstep 보간) ----
System.Func<int,int,int,double> H2 = delegate(int xi,int yi,int sd){
    unchecked {
        uint h = (uint)(xi*374761393 + yi*668265263 + sd*1442695040);
        h = (h ^ (h >> 13)) * 1274126177u; h ^= h >> 16;
        return (h & 0xFFFFFFu) / 16777216.0;
    }
};
System.Func<double,double,int,double> VN2 = delegate(double x,double y,int sd){
    int x0=(int)System.Math.Floor(x), y0=(int)System.Math.Floor(y);
    double fx=x-x0, fy=y-y0;
    double wx=fx*fx*(3.0-2.0*fx), wy=fy*fy*(3.0-2.0*fy);
    double a=H2(x0,y0,sd), b=H2(x0+1,y0,sd), c=H2(x0,y0+1,sd), d=H2(x0+1,y0+1,sd);
    return (a+(b-a)*wx) + ((c+(d-c)*wx) - (a+(b-a)*wx))*wy;
};
// 3옥타브 fBm, -1..+1
System.Func<double,double,double> FBM3 = delegate(double x,double y){
    double v = VN2(x,y,1013)*1.0 + VN2(x*2.03,y*2.03,3571)*0.5 + VN2(x*4.11,y*4.11,6151)*0.25;
    return (v/1.75)*2.0 - 1.0;
};

// ---- 크레이터풍 딤플: 중심이 어둡고 테두리가 살짝 밝다 ----
//  ★고정 목록. 난수 API 를 쓰지 않아 결정론적이다.
double[] cxs = new double[]{  92.0, 168.0, 130.0,  74.0, 186.0, 112.0, 148.0,  62.0, 200.0, 106.0 };
double[] cys = new double[]{ 104.0,  88.0, 172.0, 158.0, 150.0,  62.0, 120.0, 196.0, 118.0, 214.0 };
double[] crs = new double[]{  21.0,  14.0,  17.0,   9.0,  11.0,   7.0,  19.0,   8.0,   6.0,  12.0 };
double[] cds = new double[]{  0.22,  0.15,  0.19,  0.11,  0.13,  0.09,  0.20,  0.10,  0.08,  0.14 };

// 밝기 편차(진폭 1.0 기준)를 만드는 함수
System.Func<double,double,double> DEV = delegate(double x,double y){
    double d = FBM3(x/26.0, y/26.0) * 0.55 + FBM3(x/9.0, y/9.0) * 0.30;
    for (int k = 0; k < 10; k++) {
        double dx = x - cxs[k], dy = y - cys[k];
        double r = System.Math.Sqrt(dx*dx + dy*dy) / crs[k];
        if (r > 1.35) continue;
        double bowl = -cds[k] * (1.0 - SS(r));            // 안쪽 우묵
        double rim  =  cds[k] * 0.45 * (SS((r-0.80)/0.30) - SS((r-1.05)/0.30)); // 테두리 밝게
        d += bowl + rim;
    }
    return d;
};

// ---- 1차: 단위 진폭으로 굽고 국소 SD 측정 ----
System.Func<double,double> BUILD_SD = delegate(double amp){
    double[] v = new double[S*S];
    for (int y = 0; y < S; y++)
        for (int x = 0; x < S; x++)
            v[y*S+x] = BASE * (1.0 + amp * DEV(x, y));
    // 원판 안에서만 5x5 국소 SD / 전체 평균
    double acc = 0; int n = 0; double gm = 0; int gn = 0;
    for (int y = 2; y < S-2; y++)
        for (int x = 2; x < S-2; x++) {
            double dx = x - S*0.5, dy = y - S*0.5;
            if (System.Math.Sqrt(dx*dx + dy*dy) > DISCR - 6) continue;
            double s1 = 0, s2 = 0;
            for (int j = -2; j <= 2; j++)
                for (int i = -2; i <= 2; i++) { double q = v[(y+j)*S + (x+i)]; s1 += q; s2 += q*q; }
            double m = s1/25.0;
            acc += System.Math.Sqrt(System.Math.Max(0.0, s2/25.0 - m*m)); n++;
            gm += v[y*S+x]; gn++;
        }
    return (acc/n) / (gm/gn);
};
double sd1 = BUILD_SD(1.0);
double AMP = SDGOAL / sd1;      // SD 는 진폭에 선형

// ---- 2차: 보정된 진폭으로 최종 생성 ----
Texture2D tex = new Texture2D(S, S, TextureFormat.RGBA32, false);
Color[] outp = new Color[S*S];
for (int y = 0; y < S; y++) {
    for (int x = 0; x < S; x++) {
        double g = BASE * (1.0 + AMP * DEV(x, y));
        if (g < 0.02) g = 0.02; if (g > 1.0) g = 1.0;
        double dx = x - S*0.5 + 0.5, dy = y - S*0.5 + 0.5;
        double r = System.Math.Sqrt(dx*dx + dy*dy);
        double a = 1.0 - SS((r - (DISCR - EDGE)) / EDGE);   // 가장자리 1.5px 전이
        // ★아주 미세한 색조: 완전 무채색이면 죽은 회색으로 보인다
        outp[y*S+x] = new Color((float)(g*1.015), (float)(g*1.000), (float)(g*0.975), (float)a);
    }
}
tex.SetPixels(outp); tex.Apply();
System.IO.File.WriteAllBytes(System.IO.Path.Combine(
    System.IO.Path.GetDirectoryName(Application.dataPath), DST), tex.EncodeToPNG());
AssetDatabase.ImportAsset(DST, ImportAssetOptions.ForceUpdate);

// ---- 임포터 설정 ----
var imp = (TextureImporter)UnityEditor.AssetImporter.GetAtPath(DST);
imp.textureType = TextureImporterType.Default;
imp.sRGBTexture = true;
imp.alphaIsTransparency = true;
imp.alphaSource = TextureImporterAlphaSource.FromInput;
imp.mipmapEnabled = true;
imp.wrapMode = TextureWrapMode.Clamp;          // ★원판 밖 샘플이 0 알파 테두리로 clamp 되어야 한다
imp.filterMode = FilterMode.Bilinear;
imp.isReadable = true;
var and = new TextureImporterPlatformSettings();
and.name = "Android"; and.overridden = true; and.maxTextureSize = 256;
and.format = TextureImporterFormat.ASTC_6x6;
imp.SetPlatformTextureSettings(and);
imp.SaveAndReimport();
