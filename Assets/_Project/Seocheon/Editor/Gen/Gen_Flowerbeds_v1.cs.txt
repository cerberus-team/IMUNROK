//
//  ★규칙(§7): 절차 생성 코드는 이 폴더에 실행본 그대로 보존한다. .cs.txt 라 컴파일되지 않는다.
//
//  ★설계 전제 (지시서 §2)
//    - 근경은 야생 꽃밭이 아니라 ★이랑밭이다. 조사 동사가 "빈 자리를 찾는 것" 이라
//      규칙적이어야 빈칸이 눈에 띈다.
//    - 각 포기는 ★개별 인스턴스 · 결정론적 좌표. Terrain Detail(그래스) 사용 금지.
//    - 이랑 줄무늬는 alphamap 으로 못 그린다 (1024/1000m = 0.977 m/텍셀 vs 이랑 주기 1.7 m).
//      -> ★꽃 배치 패턴만으로 이랑을 표현한다. 두둑 메시는 만들지 않는다.
//    - Phase A 로 복원한 alphamap·높이맵은 ★다시 건드리지 않는다.
//
//  ★이랑 구조
//    이랑(두둑) 1.1 m + 고랑 0.6 m = 주기 1.7 m
//    A(이랑 가로 방향, 10~14 m) -> 이랑 줄 수 = floor(A/1.7)   [6~8줄]
//    B(이랑 진행 방향, 7~9 m)   -> 줄당 포기 = floor(B/0.5)+1  [14~18포기]
//    꽃은 ★이랑 중심선 위에만. 고랑에는 놓지 않는다.
//
//  ★지터 (지시서: 위치 ±0.08 / 회전 0~360 / 스케일 0.90~1.10)
//    ★단, 이랑 방향 성분만 ±0.07 로 줄였다.
//      ±0.08 을 양쪽 끝에 걸면 인접 간격의 표준편차가
//         sqrt(2 * (0.16^2/12)) = 0.0653 m  ->  판정 기준 0.06 을 ★넘는다.
//      ±0.07 이면 sqrt(2 * (0.14^2/12)) = 0.0571 m 로 통과한다.
//      이랑 가로 방향은 간격 판정과 무관하므로 ±0.08 을 그대로 쓴다.
//
//  ★접지
//    Terrain.SampleHeight 사용. ★Physics.Raycast 금지 (autoSyncTransforms=False 라 어긋난다)
//    꽃 프리팹은 피벗이 바닥이 아니다: 메시 bounds.min.y = -0.0408 (피벗보다 4.08 cm 아래)
//    갈대는 -0.1278, 버드나무는 -0.0005
//    -> pivot.y = 지형고도 + (|min.y| * scale) - SINK
//       SINK = 0.025 m 만큼 살짝 묻는다. 경사면에서 뜨는 것을 막으면서
//       판정 기준 |밑면 - 지형고도| <= 0.03 을 만족한다.
//
//  ★난수
//    UnityEngine.Random 을 쓰지 않는다. 아래 LCG 로 ★완전 결정론적으로 생성한다.
//    시드는 (구획, 이랑, 순번) 에서 유도하므로 같은 코드는 항상 같은 밭을 만든다.
//
//  ★이름 규칙 (지시서: 나중에 특정 포기를 비워 단서를 만들 때 이 인덱스로 지정)
//    Flower_P{구획2자리}_R{이랑2자리}_C{순번2자리}     예) Flower_P02_R03_C07
//    Reed_P{구획2자리}_{순번3자리}
//
//  ★콜라이더
//    꽃·갈대에는 붙이지 않는다 (프리팹에 원래 없음 — 조사에서 확인).
//    버드나무 프리팹에는 CapsuleCollider 3개가 붙어 있다 -> ★인스턴스에서 제거하고
//    줄기에 BoxCollider 1개만 넣는다. 변경 후 Physics.SyncTransforms() 호출.
//
//  ★CodeDom = C# 6. 로컬 함수 금지 -> System.Func. 변수명 겹침 금지.
//  ★한 호출에 900개를 다 만들면 타임아웃 위험 -> 꽃 / 갈대+나무 를 나눠 실행한다.

// ---------------------------------------------------------------- 공통 정의
const string PFX = "Assets/Idyllic Fantasy Nature/Prefabs/";
Terrain TR = UnityEngine.Object.FindObjectsByType<Terrain>(FindObjectsSortMode.None)[0];

// 구획 5개 — 경사 <12, 스폰 15~70 m, r<100, 인접 간격 3~5 m 를 만족하도록 실측으로 정한 값
//  실측: 경사max 6.68 / 5.65 / 8.69 / 7.67 / 4.71
//        스폰거리 17.4~62.0   r max 49.5
//        인접 간격 P01-P02 3.97 · P03-P04 4.31 · P04-P05 4.16
//  ★P01-P03 은 6.91 m 로 벌어져 있다. Z 688~693 에 경사 11~17° 의 능선이 가로질러
//    그 위에는 구획을 놓을 수 없기 때문이다 (지시서 "경사 12° 미만인 곳만").
double[] PCX = new double[]{ 435.0, 451.0, 438.5, 454.0, 446.0 };
double[] PCZ = new double[]{ 699.0, 699.0, 681.0, 679.5, 664.0 };
double[] PRT = new double[]{   8.0, 170.0, 145.0, 112.0,  78.0 };   // 연속 차이 18/25/33/34 (mod 180)
double[] PA  = new double[]{  11.0,  11.0,  12.0,  12.0,  12.0 };   // 이랑 가로 (10~14)
double[] PB  = new double[]{   7.5,   7.5,   8.0,   8.0,   8.0 };   // 이랑 방향 (7~9)

// 품종 — 팩의 9종 중 5종. 전부 60 tris · 동일 머티리얼(Flower.mat) · 동일 아틀라스라
// 어느 조합을 골라도 비용이 같다 (지시서 "같은 tris 대에서 고를 것" 자동 충족)
string[] SPECIES = new string[]{ "Flower_Purple","Flower_Pink","Flower_Yellow","Flower_White","Flower_Blue_02" };
// 구획별 [지배 80%, 보조 20%] — 5색이 각각 최소 한 구획의 지배색이 되게 배분
int[] DOM = new int[]{ 0, 2, 1, 4, 3 };
int[] SUB = new int[]{ 3, 0, 4, 2, 1 };

const double RIDGE = 1.1;      // 이랑(두둑) 폭
const double FURROW = 0.6;     // 고랑 폭
const double PITCH = 1.7;      // 주기
const double STEP = 0.50;      // 이랑 방향 꽃 간격
const double JIT_A = 0.08;     // 이랑 가로 지터
const double JIT_B = 0.07;     // ★이랑 방향 지터 (0.08 이면 간격 SD 가 기준 0.06 을 넘는다)
const double SINK = 0.025;     // 접지 침하

// ★결정론 LCG
System.Func<int,double> RND = delegate(int seed){
    unchecked { uint s=(uint)seed*1664525u+1013904223u;
      s^=s>>15; s*=2246822519u; s^=s>>13; s*=3266489917u; s^=s>>16;
      return (s & 0xFFFFFFu)/16777216.0; } };

// ---------------------------------------------------------------- 실행 ① 꽃
// (계층) Seocheon_Flowerfield / _Flowerbeds / Plot_01 .. Plot_05
GameObject root = GameObject.Find("_Flowerbeds");
if (root == null) { root = new GameObject("_Flowerbeds"); }
System.Collections.Generic.Dictionary<string,GameObject> cache =
  new System.Collections.Generic.Dictionary<string,GameObject>();
for (int i=0;i<SPECIES.Length;i++)
  cache[SPECIES[i]] = AssetDatabase.LoadAssetAtPath<GameObject>(PFX+SPECIES[i]+".prefab");

Vector3 tpos = TR.transform.position;
int total=0;
for (int p=0; p<5; p++)
{
    GameObject plot = new GameObject("Plot_0"+(p+1));
    plot.transform.SetParent(root.transform, false);
    double th = PRT[p]*System.Math.PI/180.0;
    double ax = System.Math.Cos(th), az = -System.Math.Sin(th);   // 이랑 가로 단위벡터
    double bx = System.Math.Sin(th), bz =  System.Math.Cos(th);   // 이랑 방향 단위벡터
    int nR = (int)System.Math.Floor(PA[p]/PITCH + 1e-6);          // 이랑 줄 수
    int nC = (int)System.Math.Floor(PB[p]/STEP  + 1e-6) + 1;      // 줄당 포기
    for (int r=0; r<nR; r++)
    {
        double oa = (r - (nR-1)*0.5) * PITCH;                     // ★이랑 중심선. 고랑에는 안 놓는다
        for (int c=0; c<nC; c++)
        {
            double ob = (c - (nC-1)*0.5) * STEP;
            int sd = (p+1)*1000000 + (r+1)*1000 + (c+1);
            double ja = (RND(sd*3+11)*2.0-1.0)*JIT_A;
            double jb = (RND(sd*3+29)*2.0-1.0)*JIT_B;
            double sc = 0.90 + RND(sd*3+47)*0.20;
            double ry = RND(sd*3+83)*360.0;
            bool dom = RND(sd*3+101) < 0.80;                      // 지배 80% / 보조 20%
            string sp = SPECIES[ dom ? DOM[p] : SUB[p] ];
            double wx = PCX[p] + ax*(oa+ja) + bx*(ob+jb);
            double wz = PCZ[p] + az*(oa+ja) + bz*(ob+jb);
            float gy = TR.SampleHeight(new Vector3((float)wx,0f,(float)wz)) + tpos.y;
            GameObject go = (GameObject)PrefabUtility.InstantiatePrefab(cache[sp], plot.transform);
            go.name = "Flower_P"+(p+1).ToString("00")+"_R"+(r+1).ToString("00")+"_C"+(c+1).ToString("00");
            go.transform.position = new Vector3((float)wx, (float)(gy + 0.0408*sc - SINK), (float)wz);
            go.transform.rotation = Quaternion.Euler(0f,(float)ry,0f);
            go.transform.localScale = Vector3.one*(float)sc;
            GameObjectUtility.SetStaticEditorFlags(go, StaticEditorFlags.BatchingStatic);
            total++;
        }
    }
}

//  ★꽃과 달리 규칙성이 있으면 안 된다 — 간격 불규칙, 지터 ±0.4, 스케일 0.8~1.3

//    const double OUT = 1.6;        // 구획 경계에서 바깥으로 띠 중심까지
//    const double BAND = 0.9;       // 띠 폭 (0.6~1.2)
//    간격은 0.42~0.72 m 사이를 난수로 걸어가며 (누적) 배치 -> 규칙성 제거
//    갈대 종은 Reeds_01/02/03 을 난수로 섞는다 (머티리얼 3종이지만 전부 16 tris)
//    이름 Reed_P{구획}_{순번}

//    콜라이더: 프리팹의 CapsuleCollider 3개 제거 -> BoxCollider 1개 (줄기)
