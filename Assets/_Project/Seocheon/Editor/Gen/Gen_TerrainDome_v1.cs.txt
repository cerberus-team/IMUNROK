  밭을 동산에 올리면 사면이 관측자 쪽으로 얼굴을 돌려 면적이 생긴다.

[2] 적용식

  r    = |(wx,wz) - (FCX,FCZ)|                      밭 중심으로부터 수평 거리
  dF   = 밭 영역까지의 외부거리 (밭 안이면 0)
  u    = clamp(dF / FADE, 0, 1)
  s    = u*u*(3 - 2*u)                              ★GLSL smoothstep 직접 구현
                                                    (Mathf.SmoothStep 은 인자 의미가 다름)
  dome = A_DOME * exp( -(r/Rd)^2 ) * (1 - s)
  H_new = H_orig + dome / 300.0                     ★높이맵은 0~1 정규화, size.y = 300

  -> 돔 자체가 가우시안으로 감쇠하고, 페이드 마스크가 밭 밖 16 m 에서 원지형으로 되돌린다
  -> 밭 경계(dF=0)에서 돔은 이미 50% 수준이라 봉우리가 밭 안에 갇히지 않는다

  적용 후  ★td.SyncHeightmap()  →  ★Physics.SyncTransforms()

[7] 실행 코드 (UnityMCP execute_code · C# 6 · 로컬함수 금지 · UnityEngine.Object 명시)

var T = UnityEngine.Object.FindFirstObjectByType<Terrain>();
var td = T.terrainData;
int HR = td.heightmapResolution;                  // 513
Vector3 SZ = td.size, TP = T.transform.position;  // (1000,300,1000) / (0,0,0)
float TEX = SZ.x / (HR - 1);                      // 1.953125

const float A_DOME = 3.50f, FADE = 16f, MARGIN = 2f, PATHHW = 1.5f;
const float FCX = 447.67f, FCZ = 679.15f;
float Rd = 22.05f;

float[,] H0 = td.GetHeights(0, 0, HR, HR);        // ★원본 보관

float[] CX = {434.000f,446.990f,438.842f,449.801f,460.399f,438.114f,450.737f,462.507f};
float[] CZ = {699.600f,698.638f,678.888f,677.770f,677.291f,666.050f,667.264f,667.737f};
float[] RO = {8f,172f,145f,112f,78f,45f,20f,165f};
float[] SA = {11f,11f,9f,9f,9f,9f,9f,9f};
int[] PA = {0,2,3,5,6,2,3,4};
int[] PB = {1,3,4,6,7,5,6,7};
float[] AXX = new float[8], AXZ = new float[8], BXX = new float[8], BXZ = new float[8];
for (int b = 0; b < 8; b++) {
    float th = RO[b] * Mathf.Deg2Rad;
    AXX[b] =  Mathf.Cos(th); AXZ[b] = -Mathf.Sin(th);
    BXX[b] =  Mathf.Sin(th); BXZ[b] =  Mathf.Cos(th);
}

float[,] HN = new float[HR, HR];
for (int z = 0; z < HR; z++) {
    float wz = TP.z + z * TEX;
    for (int x = 0; x < HR; x++) {
        float wx = TP.x + x * TEX;
        float dF = 1e9f;
        for (int b = 0; b < 8; b++) {
            float ox = wx - CX[b], oz = wz - CZ[b];
            float la = ox * AXX[b] + oz * AXZ[b];
            float lb = ox * BXX[b] + oz * BXZ[b];
            float qa = Mathf.Max(Mathf.Abs(la) - (SA[b] * 0.5f + MARGIN), 0f);
            float qb = Mathf.Max(Mathf.Abs(lb) - 5.5f, 0f);
            float dr = Mathf.Sqrt(qa * qa + qb * qb);
            if (dr < dF) dF = dr;
        }
        for (int p = 0; p < 8; p++) {
            float px = CX[PA[p]], pz = CZ[PA[p]];
            float ex = CX[PB[p]] - px, ez = CZ[PB[p]] - pz;
            float tt = Mathf.Clamp01(((wx - px) * ex + (wz - pz) * ez) / (ex * ex + ez * ez));
            float cx2 = px + tt * ex, cz2 = pz + tt * ez;
            float ds = Mathf.Sqrt((wx - cx2) * (wx - cx2) + (wz - cz2) * (wz - cz2));
            float dp = Mathf.Max(ds - PATHHW, 0f);
            if (dp < dF) dF = dp;
        }
        float r = Mathf.Sqrt((wx - FCX) * (wx - FCX) + (wz - FCZ) * (wz - FCZ));
        float u = Mathf.Clamp01(dF / FADE);
        float s = u * u * (3f - 2f * u);                    // ★GLSL smoothstep
        float dome = A_DOME * Mathf.Exp(-(r / Rd) * (r / Rd)) * (1f - s);
        HN[z, x] = H0[z, x] + dome / SZ.y;                  // ★1 m = 1/300
    }
}
td.SetHeights(0, 0, HN);
td.SyncHeightmap();
Physics.SyncTransforms();
