[3] ★smoothstep — GLSL 의미로 직접 구현

  ★Mathf.SmoothStep(from, to, t) 은 인자 의미가 다르다. 쓰지 말 것.

     float u = Mathf.Clamp01( (x - e0) / (e1 - e0) );
     float s = u * u * (3f - 2f * u);

  적용
     s = smoothstep(0, FADE, dFarm)
     hNew = g * (1 - s) + h0 * s
       dFarm = 0     -> s = 0 -> hNew = H_target     (밭 안)
       dFarm >= FADE -> s = 1 -> hNew = h0           (밖, 완전 원지형)

[7] 실행 코드 (UnityMCP execute_code · C# 6 · 로컬함수 금지 · UnityEngine.Object 명시)

var T = UnityEngine.Object.FindFirstObjectByType<Terrain>();
var td = T.terrainData;
int HR = td.heightmapResolution;                 // 513
Vector3 SZ = td.size;                            // (1000,300,1000)
Vector3 TP = T.transform.position;               // (0,0,0)
float TEX = SZ.x / (HR - 1);                     // 1.953125

const float SIGMA = 12f, MARGIN = 2f, PATHHW = 1.5f, FADE = 10f;
int KR = 19;

float[,] h0 = td.GetHeights(0, 0, HR, HR);       // h0[z, x]

// --- 1D 가우시안 가중치 ---
float[] wk = new float[2 * KR + 1];
float wsum = 0f;
for (int k = -KR; k <= KR; k++) {
    float d = k * TEX;
    float v = Mathf.Exp(-(d * d) / (2f * SIGMA * SIGMA));
    wk[k + KR] = v; wsum += v;
}
for (int i = 0; i < wk.Length; i++) wk[i] /= wsum;

// --- 분리형 평활 : x 방향 ---
float[,] gx = new float[HR, HR];
for (int z = 0; z < HR; z++)
    for (int x = 0; x < HR; x++) {
        float a = 0f;
        for (int k = -KR; k <= KR; k++) {
            int xx = Mathf.Clamp(x + k, 0, HR - 1);
            a += wk[k + KR] * h0[z, xx];
        }
        gx[z, x] = a;
    }
// --- 분리형 평활 : z 방향 ---
float[,] g = new float[HR, HR];
for (int z = 0; z < HR; z++)
    for (int x = 0; x < HR; x++) {
        float a = 0f;
        for (int k = -KR; k <= KR; k++) {
            int zz = Mathf.Clamp(z + k, 0, HR - 1);
            a += wk[k + KR] * gx[zz, x];
        }
        g[z, x] = a;
    }

// --- 배치 ---
float[] CX = {434.000f,446.990f,438.842f,449.801f,460.399f,438.114f,450.737f,462.507f};
float[] CZ = {699.600f,698.638f,678.888f,677.770f,677.291f,666.050f,667.264f,667.737f};
float[] RT = {  8.0f, 172.0f, 145.0f, 112.0f,  78.0f,  45.0f,  20.0f, 165.0f};
float[] SA = { 11.0f,  11.0f,   9.0f,   9.0f,   9.0f,   9.0f,   9.0f,   9.0f};
float[] SB = {  7.0f,   7.0f,   7.0f,   7.0f,   7.0f,   7.0f,   7.0f,   7.0f};
int[]   PA = {0,2,3,5,6,2,3,4};
int[]   PB = {1,3,4,6,7,5,6,7};

float[] AXX = new float[8], AXZ = new float[8], BXX = new float[8], BXZ = new float[8];
for (int b = 0; b < 8; b++) {
    float th = RT[b] * Mathf.Deg2Rad;
    AXX[b] =  Mathf.Cos(th); AXZ[b] = -Mathf.Sin(th);
    BXX[b] =  Mathf.Sin(th); BXZ[b] =  Mathf.Cos(th);
}

// --- 적용 ---
float[,] hN = new float[HR, HR];
float maxAbs = 0f; int over15 = 0; float spawnMax = 0f;
int mxx = -1, mzz = -1;

for (int z = 0; z < HR; z++) {
    float wz = TP.z + z * TEX;
    for (int x = 0; x < HR; x++) {
        float wx = TP.x + x * TEX;

        float dFarm = 1e9f;
        for (int b = 0; b < 8; b++) {
            float ox = wx - CX[b], oz = wz - CZ[b];
            float la = ox * AXX[b] + oz * AXZ[b];
            float lb = ox * BXX[b] + oz * BXZ[b];
            float qa = Mathf.Max(Mathf.Abs(la) - (SA[b] * 0.5f + MARGIN), 0f);
            float qb = Mathf.Max(Mathf.Abs(lb) - (SB[b] * 0.5f + MARGIN), 0f);
            float dr = Mathf.Sqrt(qa * qa + qb * qb);
            if (dr < dFarm) dFarm = dr;
        }
        for (int p = 0; p < 8; p++) {
            float px = CX[PA[p]], pz = CZ[PA[p]];
            float ex = CX[PB[p]] - px, ez = CZ[PB[p]] - pz;
            float L2 = ex * ex + ez * ez;
            float tt = Mathf.Clamp01(((wx - px) * ex + (wz - pz) * ez) / L2);
            float cx2 = px + tt * ex, cz2 = pz + tt * ez;
            float ds = Mathf.Sqrt((wx - cx2) * (wx - cx2) + (wz - cz2) * (wz - cz2));
            float dp = Mathf.Max(ds - PATHHW, 0f);
            if (dp < dFarm) dFarm = dp;
        }

        // ★GLSL smoothstep(0, FADE, dFarm)
        float u = Mathf.Clamp01(dFarm / FADE);
        float s = u * u * (3f - 2f * u);

        float nv = g[z, x] * (1f - s) + h0[z, x] * s;
        hN[z, x] = nv;

        float dm = (nv - h0[z, x]) * SZ.y;         // ★m 단위
        float ad = Mathf.Abs(dm);
        if (ad > maxAbs) { maxAbs = ad; mxx = x; mzz = z; }
        if (ad > 1.5f) over15++;
        float sdx = wx - 444.1f, sdz = wz - 719.3f;
        if (sdx * sdx + sdz * sdz <= 9f && ad > spawnMax) spawnMax = ad;
