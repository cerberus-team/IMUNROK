System.Func<string, Vector3, Vector3, uint, Color, GameObject> makeCloud = delegate(string n, Vector3 pos, Vector3 box, uint seed, Color tint)
{
    var go = new GameObject(n); go.transform.SetParent(clouds.transform, false); go.transform.position = pos;
    var ps = go.AddComponent<ParticleSystem>(); ps.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear); ps.useAutoRandomSeed = false; ps.randomSeed = seed;
    var m = ps.main; m.loop = true; m.prewarm = true; m.duration = 40f; m.simulationSpace = ParticleSystemSimulationSpace.World; m.maxParticles = 28;
    m.startLifetime = new ParticleSystem.MinMaxCurve(100f, 150f); m.startSpeed = new ParticleSystem.MinMaxCurve(.015f, .06f); m.startSize = new ParticleSystem.MinMaxCurve(42f, 88f); m.startColor = tint; m.gravityModifier = 0f;
    var em = ps.emission; em.rateOverTime = .08f; em.SetBursts(new ParticleSystem.Burst[] { new ParticleSystem.Burst(0f, 18) });
    var sh = ps.shape; sh.enabled = true; sh.shapeType = ParticleSystemShapeType.Box; sh.scale = box;
    var no = ps.noise; no.enabled = true; no.quality = ParticleSystemNoiseQuality.Low; no.strength = .12f; no.frequency = .035f; no.scrollSpeed = .01f;
    var co = ps.colorOverLifetime; co.enabled = true; Gradient g = new Gradient();
    g.SetKeys(new GradientColorKey[] { new GradientColorKey(Color.white, 0f), new GradientColorKey(Color.white, 1f) }, new GradientAlphaKey[] { new GradientAlphaKey(0f, 0f), new GradientAlphaKey(1f, .12f), new GradientAlphaKey(.92f, .85f), new GradientAlphaKey(0f, 1f) }); co.color = g;
    var r = go.GetComponent<ParticleSystemRenderer>(); r.renderMode = ParticleSystemRenderMode.Billboard; r.alignment = ParticleSystemRenderSpace.View; r.sharedMaterial = cloudMat; r.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off; r.receiveShadows = false;
    ps.Play(); ps.Simulate(48f, true, true, false); return go;
};

System.Func<string, Vector3, Vector3, uint, GameObject> makeAurora = delegate(string n, Vector3 pos, Vector3 box, uint seed)
{
    var go = new GameObject(n); go.transform.SetParent(auroras.transform, false); go.transform.position = pos;
    var ps = go.AddComponent<ParticleSystem>(); ps.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear); ps.useAutoRandomSeed = false; ps.randomSeed = seed;
    var m = ps.main; m.loop = true; m.prewarm = true; m.duration = 45f; m.simulationSpace = ParticleSystemSimulationSpace.World; m.maxParticles = 18;
    m.startLifetime = new ParticleSystem.MinMaxCurve(120f, 180f); m.startSpeed = new ParticleSystem.MinMaxCurve(.015f, .045f); m.startSize3D = true;
    m.startSizeX = new ParticleSystem.MinMaxCurve(16f, 30f); m.startSizeY = new ParticleSystem.MinMaxCurve(90f, 145f); m.startSizeZ = 1f;
    m.startColor = new ParticleSystem.MinMaxGradient(new Color(1f, .16f, .78f, .12f), new Color(.12f, 1f, .86f, .16f));
    var em = ps.emission; em.rateOverTime = .025f; em.SetBursts(new ParticleSystem.Burst[] { new ParticleSystem.Burst(0f, 12) });
    var sh = ps.shape; sh.enabled = true; sh.shapeType = ParticleSystemShapeType.Box; sh.scale = box;
    var no = ps.noise; no.enabled = true; no.quality = ParticleSystemNoiseQuality.Low; no.strength = .10f; no.frequency = .025f; no.scrollSpeed = .008f;
    var r = go.GetComponent<ParticleSystemRenderer>(); r.renderMode = ParticleSystemRenderMode.Billboard; r.alignment = ParticleSystemRenderSpace.View; r.sharedMaterial = auroraMat; r.shadowCastingMode = UnityEngine.Rendering.ShadowCastingMode.Off; r.receiveShadows = false;
    ps.Play(); ps.Simulate(60f, true, true, false); return go;
};
