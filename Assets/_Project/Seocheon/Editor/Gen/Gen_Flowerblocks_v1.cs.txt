//  ★결합 메시의 함정 — 바람 셰이더

//     굽힘이 ★로컬 y 를 기준으로 걸리므로, 결합 오브젝트를 원점에 두면 모든 정점의 로컬 y 가
//     10 m 근처가 되어 ★전체가 최대로 휘어 버린다.
//     -> 결합 오브젝트의 트랜스폼을 ★구역 중심(지형 고도 포함)에 두고,
//        결합 행렬을 parent.worldToLocalMatrix * instance.localToWorldMatrix 로 만들어
//        로컬 y 를 원래 포기 높이 범위(-0.5 ~ +1.0)로 유지한다.
//     -> 그래도 이상하면 그 구역만 개별 인스턴스로 되돌리고 보고할 것.
//
//  ★접지  Terrain.SampleHeight (Physics.Raycast 금지 — autoSyncTransforms=False)
//         pivot.y = 지형고도 + 0.0408*scale - 0.025(침하)   [지난 라운드 값 유지]
//  ★꽃·갈대에 콜라이더를 붙이지 않는다 (프리팹에 원래 없음)
//  ★난수는 UnityEngine.Random 을 쓰지 않는다. 아래 LCG 로 완전 결정론적.

const string PFX = "Assets/Idyllic Fantasy Nature/Prefabs/";
const double GRID = 0.31;     // 격자 간격
const double JIT  = 0.06;     // 위치 지터
const double SINK = 0.025;    // 접지 침하
const double PIV  = 0.0408;   // 꽃 피벗-바닥 오프셋

double[] CX=new double[]{434.000,446.990,438.842,449.801,460.399,438.114,450.737,462.507};
double[] CZ=new double[]{699.600,698.638,678.888,677.770,677.291,666.050,667.264,667.737};
double[] RT=new double[]{  8.0, 172.0, 145.0, 112.0,  78.0,  45.0,  20.0, 165.0};
double[] SA=new double[]{ 11.0,  11.0,   9.0,   9.0,   9.0,   9.0,   9.0,   9.0};
double[] SB=new double[]{  7.0,   7.0,   7.0,   7.0,   7.0,   7.0,   7.0,   7.0};
string[] SPEC=new string[]{"Flower_Purple","Flower_Yellow","Flower_Pink","Flower_Blue_02",
                           "Flower_White","Flower_Blue_02","Flower_Yellow","Flower_Pink"};

System.Func<int,double> RND = delegate(int seed){
    unchecked { uint s=(uint)seed*1664525u+1013904223u;
      s^=s>>15; s*=2246822519u; s^=s>>13; s*=3266489917u; s^=s>>16;
      return (s & 0xFFFFFFu)/16777216.0; } };

// ---------------------------------------------------------------- 실행 ① 근접군 (B01·B02)
//   각 포기 개별 GameObject · BatchingStatic · 이름 Flower_B##_R##_C##
//   for b in {0,1}:
//     th = RT[b] deg;  ax=(cos th, -sin th)  이랑 가로;  bx=(sin th, cos th)
//     nR = floor(SA/GRID)+1 = 36,  nC = floor(SB/GRID)+1 = 23
//     oa = (r-(nR-1)/2)*GRID,  ob = (c-(nC-1)/2)*GRID
//     지터 ja,jb = (RND*2-1)*JIT,  scale 0.90+RND*0.20,  yaw RND*360
//     y = Terrain.SampleHeight(x,z) + terrainPos.y + PIV*scale - SINK

// ---------------------------------------------------------------- 실행 ② 원경군 (B03~B08)
