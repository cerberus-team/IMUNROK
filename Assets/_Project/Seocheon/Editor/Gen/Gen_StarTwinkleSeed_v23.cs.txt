//  대상  Assets/_Project/Seocheon/Art/Sky/M_StarBillboards.asset  (별 900개 단일 메시)
//  내용  uv4.x (= 셰이더의 v.rw.x, 반짝임 진폭 시드) 의 분포 하한만 올린다
//        0.20 ~ 1.00  ->  0.35 ~ 1.00
//
//  ★메시를 다시 "생성" 하지 않고 ★기존 메시의 한 속성만 선형 재사상한다.
//    재생성하면 위치·위상·원속도·가중·크기·등급이 전부 새 난수가 되어
//    이미 통과한 지표(주기 1.5~3.9초 · 위상 흩어짐 · 화면 점유율 0.564% · DC 1)가
//    전부 다시 흔들린다. 재사상은 그 위험이 0 이다.
//
//  ★건드리지 않는 것
//    uv4.y (가중 시드 w1) · uv2 (위상 p1 · 원속도) · uv3 (center) · color (등급)
//    정점 위치 · 삼각형 · 별 개수 900 · 서브메시 1개
//    _TwinkleDepth 0.42/0.35 · _TwinkleZenith 0.40 · s1/s2 수식
//
//  ★CodeDom = C# 6. 로컬 함수 금지. 한글 식별자 금지.
// ============================================================================

const string MESH = "Assets/_Project/Seocheon/Art/Sky/M_StarBillboards.asset";
const double OLD_LO = 0.20;   // 현재 분포 하한 (실측 0.2000)
const double OLD_HI = 1.00;   // 현재 분포 상한 (실측 0.9989)
const double NEW_LO = 0.35;   // 목표 하한
const double NEW_HI = 1.00;   // 목표 상한 (무변경)

Mesh mesh = AssetDatabase.LoadAssetAtPath<Mesh>(MESH);

var rw = new System.Collections.Generic.List<Vector2>();
mesh.GetUVs(3, rw);           // TEXCOORD3 = uv4 = float2 rw

double scale = (NEW_HI - NEW_LO) / (OLD_HI - OLD_LO);
for (int i = 0; i < rw.Count; i++)
{
    double v = rw[i].x;
    double n = NEW_LO + (v - OLD_LO) * scale;
    if (n < NEW_LO) n = NEW_LO;
    if (n > NEW_HI) n = NEW_HI;
    rw[i] = new Vector2((float)n, rw[i].y);   // ★y(가중 시드)는 그대로
}
mesh.SetUVs(3, rw);
mesh.UploadMeshData(false);   // ★false : 이후에도 읽을 수 있게 CPU 사본 유지
EditorUtility.SetDirty(mesh);
AssetDatabase.SaveAssets();
