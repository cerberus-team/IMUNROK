string[] names={"S31_BEFORE_P1.png","S31_FINAL_P1.png","S31_BEFORE_BACK.png","S31_FINAL_BACK.png","S31_BEFORE_BASIN.png","S31_FINAL_BASIN.png"};System.Text.StringBuilder sb=new System.Text.StringBuilder();
for(int n=0;n<names.Length;n++){string p=System.IO.Path.Combine(Application.dataPath,"Screenshots/"+names[n]);Texture2D t=new Texture2D(2,2,TextureFormat.RGB24,false);t.LoadImage(System.IO.File.ReadAllBytes(p));Color32[] px=t.GetPixels32();double sat=0,lum=0,skyLum=0;long cnt=0,clip=0,skyCnt=0;for(int y=0;y<t.height;y++)for(int x=0;x<t.width;x++){Color c=px[y*t.width+x];float h,s,v;Color.RGBToHSV(c,out h,out s,out v);double l=0.2126*c.r+0.7152*c.g+0.0722*c.b;if(y<t.height/2){sat+=s;lum+=l;cnt++;if(c.r>=0.98f||c.g>=0.98f||c.b>=0.98f)clip++;}else{skyLum+=l;skyCnt++;}}sb.Append(names[n]).Append("|groundSat=").Append((sat/cnt).ToString("F4")).Append("|groundLum=").Append((lum/cnt).ToString("F4")).Append("|clip=").Append((100.0*clip/cnt).ToString("F2")).Append("%|upperLum=").Append((skyLum/skyCnt).ToString("F4")).Append("\n");UnityEngine.Object.DestroyImmediate(t);}
return sb.ToString();
